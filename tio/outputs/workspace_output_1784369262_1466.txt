===== SHOP.JS =====
/**
 * js/pages/shop.js
 * Shop page logic:
 *   - loads /data/shops.json (static demo data)
 *   - renders a tab per shop (Red Label, Blue House, Yellow Studio)
 *   - shows the active shop's header (name, tagline, colour dot)
 *   - renders each item as a card with a colour preview, name and price
 *
 * No checkout — this is a proof-of-concept browsing experience.
 */

'use strict';

(function () {
  // Category emoji icons (used because items are colour-only, no photos).
  const CATEGORY_ICON = { shirt: '👕', pants: '👖', hat: '🧢' };
  const COLOR_LABEL   = { red: 'Red', blue: 'Blue', yellow: 'Yellow' };

  let shops = [];
  let activeId = null;

  document.addEventListener('DOMContentLoaded', async () => {
    try {
      const data = await Api.getShops();
      shops = data.shops || [];
    } catch (_) {
      document.getElementById('shopItems').innerHTML =
        '<p class="text-muted">Could not load shops.</p>';
      return;
    }

    if (shops.length === 0) return;

    activeId = shops[0].id;
    renderTabs();
    renderShop(activeId);
  });

  /* ---------- Tabs ---------- */
  function renderTabs() {
    const host = document.getElementById('shopTabs');
    if (!host) return;
    host.innerHTML = shops
      .map((s) => `<button class="shop-tab${s.id === activeId ? ' is-active' : ''}" data-id="${s.id}">${Utils.escape(s.name.split('—')[0].trim())}</button>`)
      .join('');

    host.addEventListener('click', (e) => {
      const tab = e.target.closest('.shop-tab');
      if (!tab) return;
      activeId = tab.dataset.id;
      host.querySelectorAll('.shop-tab').forEach((t) => t.classList.toggle('is-active', t.dataset.id === activeId));
      renderShop(activeId);
    });
  }

  /* ---------- Active shop ---------- */
  function renderShop(id) {
    const shop = shops.find((s) => s.id === id);
    if (!shop) return;

    // Banner.
    const banner = document.getElementById('shopBanner');
    if (banner) banner.hidden = false;
    const name = document.getElementById('shopName');
    if (name) name.textContent = shop.name;
    const tagline = document.getElementById('shopTagline');
    if (tagline) tagline.textContent = shop.tagline;
    const dot = document.getElementById('shopColorDot');
    if (dot) {
      dot.className = `swatch swatch--${shop.color}`;
      dot.title = COLOR_LABEL[shop.color];
    }

    // Items.
    const itemsHost = document.getElementById('shopItems');
    if (!itemsHost) return;
    itemsHost.innerHTML = shop.items.map(renderItem).join('');
  }

  function renderItem(item) {
    const bgVar = `var(--color-${item.color})`;
    const soft  = `rgba(${item.color === 'red' ? '230,57,70' : item.color === 'blue' ? '42,109,244' : '244,194,13'},0.10)`;
    return `
      <div class="shop-item">
        <div class="shop-item__img" style="background:${soft}">
          <span style="font-size:48px">${CATEGORY_ICON[item.category] || '🛍️'}</span>
        </div>
        <div class="shop-item__name">${Utils.escape(item.name)}</div>
        <div class="shop-item__price">$${item.price}</div>
        <div class="flex" style="justify-content:center;gap:var(--space-2)">
          <span class="badge badge--${item.color}">${COLOR_LABEL[item.color]}</span>
          <span class="badge">${item.category}</span>
        </div>
      </div>`;
  }
})();
