/**
 * js/pages/avatar.js
 * Avatar Designer page logic.
 *
 * Renders a live 2D avatar and lets the user swap the colour of each
 * clothing layer (shirt / pants / hat) via swatch buttons. A
 * "Randomise" button shuffles all three colours at once.
 *
 * The chosen colours are persisted to localStorage so they become the
 * defaults when the user navigates to the Outfit Designer.
 */

'use strict';

(function () {
  const COLORS = Avatar.colors();

  // Initial state — restored from localStorage if available.
  const state = restoreState({
    shirt: 'red',
    pants: 'blue',
    hat: 'yellow'
  });

  document.addEventListener('DOMContentLoaded', () => {
    const preview = document.getElementById('avatarPreview');
    if (!preview) return;

    // Render the avatar once.
    preview.innerHTML = Avatar.render(state, 'lg');
    updateReadout();

    // Delegate clicks on every swatch group.
    document.querySelectorAll('.swatch-row').forEach((group) => {
      const layer = group.dataset.layer;
      group.addEventListener('click', (e) => {
        const btn = e.target.closest('.swatch-btn');
        if (!btn) return;
        selectSwatch(group, btn);
        state[layer] = btn.dataset.color;
        Avatar.setColor(preview, layer, btn.dataset.color);
        updateReadout();
        saveState();
      });
    });

    // Randomise button.
    const randomBtn = document.getElementById('randomBtn');
    if (randomBtn) {
      randomBtn.addEventListener('click', () => {
        COLORS.forEach((layer) => randomiseLayer(layer));
        updateReadout();
        saveState();
      });
    }
  });

  /* ---------- Helpers ---------- */

  /** Mark a single swatch as selected within its group. */
  function selectSwatch(group, btn) {
    group.querySelectorAll('.swatch-btn').forEach((b) => b.classList.remove('is-selected'));
    btn.classList.add('is-selected');
  }

  /** Update the colour-readout badges under the avatar. */
  function updateReadout() {
    const cap = (s) => s.charAt(0).toUpperCase() + s.slice(1);
    const set = (id, label) => {
      const el = document.getElementById(id);
      if (el) el.textContent = `${cap(state[label])} ${label}`;
    };
    set('readShirt', 'shirt');
    set('readPants', 'pants');
    set('readHat', 'hat');
  }

  /** Pick a random colour for one layer and sync the UI. */
  function randomiseLayer(layer) {
    const color = COLORS[Math.floor(Math.random() * COLORS.length)];
    state[layer] = color;

    // Update the swatch buttons.
    const group = document.querySelector(`.swatch-row[data-layer="${layer}"]`);
    if (group) {
      const btn = group.querySelector(`[data-color="${color}"]`);
      if (btn) selectSwatch(group, btn);
    }

    // Update the avatar preview.
    const preview = document.getElementById('avatarPreview');
    if (preview) Avatar.setColor(preview, layer, color);
  }

  /** Persist the avatar colours for use on the Outfit Designer page. */
  function saveState() {
    try {
      localStorage.setItem('tryiton.avatar', JSON.stringify(state));
    } catch (_) { /* storage may be unavailable */ }
  }

  /** Restore saved colours, falling back to defaults. */
  function restoreState(defaults) {
    try {
      const raw = localStorage.getItem('tryiton.avatar');
      if (raw) {
        const parsed = JSON.parse(raw);
        return {
          shirt: COLORS.includes(parsed.shirt) ? parsed.shirt : defaults.shirt,
          pants: COLORS.includes(parsed.pants) ? parsed.pants : defaults.pants,
          hat:   COLORS.includes(parsed.hat)   ? parsed.hat   : defaults.hat
        };
      }
    } catch (_) { /* ignore */ }
    return defaults;
  }
})();
