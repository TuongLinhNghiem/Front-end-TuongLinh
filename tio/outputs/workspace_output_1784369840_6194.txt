     1	/**
     2	 * controllers/outfitController.js
     3	 * CRUD operations for outfits belonging to the logged-in user.
     4	 *
     5	 * All endpoints are JSON (consumed by the frontend via fetch()).
     6	 */
     7	
     8	'use strict';
     9	
    10	const Outfit = require('../models/outfit');
    11	const validate = require('../middleware/validate');
    12	
    13	const outfitController = {
    14	  /** GET /api/outfits — list the current user's outfits. */
    15	  async list(req, res) {
    16	    const outfits = await Outfit.findByUser(req.session.user.id);
    17	    res.json(outfits);
    18	  },
    19	
    20	  /** POST /api/outfits — create a new outfit. */
    21	  async create(req, res) {
    22	    const errors = validate.outfit(req.body);
    23	    if (errors.length) return res.status(400).json({ error: errors.join(' | ') });
    24	
    25	    const outfit = await Outfit.create({
    26	      user_id: req.session.user.id,
    27	      name: req.body.name,
    28	      shirt: req.body.shirt,
    29	      pants: req.body.pants,
    30	      hat: req.body.hat
    31	    });
    32	    res.status(201).json(outfit);
    33	  },
    34	
    35	  /** PUT /api/outfits/:id — update an existing outfit. */
    36	  async update(req, res) {
    37	    const existing = await Outfit.findById(req.params.id);
    38	    if (!existing || existing.user_id !== req.session.user.id) {
    39	      return res.status(404).json({ error: 'Outfit not found.' });
    40	    }
    41	    const errors = validate.outfit(req.body);
    42	    if (errors.length) return res.status(400).json({ error: errors.join(' | ') });
    43	
    44	    const updated = await Outfit.update(req.params.id, req.body);
    45	    res.json(updated);
    46	  },
    47	
    48	  /** DELETE /api/outfits/:id — remove an outfit. */
    49	  async remove(req, res) {
    50	    const existing = await Outfit.findById(req.params.id);
    51	    if (!existing || existing.user_id !== req.session.user.id) {
    52	      return res.status(404).json({ error: 'Outfit not found.' });
    53	    }
    54	    await Outfit.remove(req.params.id);
    55	    // 204 No Content — standard for a successful DELETE.
    56	    res.status(204).end();
    57	  },
    58	
    59	  /** POST /api/outfits/:id/duplicate — clone an outfit with a "(copy)" suffix. */
    60	  async duplicate(req, res) {
    61	    const existing = await Outfit.findById(req.params.id);
    62	    if (!existing || existing.user_id !== req.session.user.id) {
    63	      return res.status(404).json({ error: 'Outfit not found.' });
    64	    }
    65	    const copy = await Outfit.create({
    66	      user_id: existing.user_id,
    67	      name: `${existing.name} (copy)`,
    68	      shirt: existing.shirt,
    69	      pants: existing.pants,
    70	      hat: existing.hat
    71	    });
    72	    res.status(201).json(copy);
    73	  }
    74	};
    75	
    76	module.exports = outfitController;
